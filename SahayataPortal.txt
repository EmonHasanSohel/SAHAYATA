// T-01: User Registration Module Structure

struct UserRegistration {
    char name[100];
    char email[100];
    char username[50];
    char password[50];
    char phoneNumber[20];
    char address[200];
    
    // Verification Flags
    bool isOtpVerified;
    bool isEmailConfirmed;
};

// T-02: Login and Forgot Password Module Structure

struct UserLogin {
    char username[50];
    char password[50];
    char userRole[20]; // Victim, Donator, Volunteer, Admin, Manager
    
    // Forgot Password Flags
    bool isOtpRequested;
    char resetOtp[10];
    char newPassword[50];
};
 
// T-03: Manager Profile Management Structure

struct ManagerProfile {
    char name[100];
    char contactNumber[20];
    char email[100];
    char address[200];
    char accountStatus[20]; // Active or Inactive
    
    // Profile Update Flags
    bool isProfileUpdated;
};

// T-04: Admin Account Management by Manager

struct AdminAccountManagement {
    char adminName[100];
    char adminEmail[100];
    char adminContact[20];
    char creationDate[20];
    char accountStatus[20]; // Active or Inactive
    
    // Management Actions
    bool canCreateAdmin;
    bool canActivateAdmin;
    bool canDeactivateAdmin;
};

// T-05: System Statistics Dashboard for Manager

struct SystemStatistics {

    int totalRegisteredDonators;
    int totalRegisteredVolunteers;
    int totalActiveAdmins;
    
    int totalDisasterRequests;
    int totalDonationItems;
    int totalDeliveriesCompleted;
    
    // Dashboard Refresh
    bool isDataRealTime;
};

// T-07: Admin Disaster Request Management Structure

struct DisasterRequestManagement {
    char requestId[20];
    char victimId[20];
    char disasterType[50];
    char location[100];
    char urgencyLevel[20]; // High, Medium, Low
    char requestStatus[20]; // Pending, Approved, Rejected
    
    // Admin Actions
    bool canSearchRequests;
    bool canFilterRequests;
    char actionReason[200]; // Optional reason for approval/rejection
};
